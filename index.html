EX. 1

UNIX System Calls — fork, exec, getpid, wait, exit, close, stat, opendir, readdir

CCompile: gcc program.c -o program && ./program
#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <dirent.h>
#include <string.h>
#include <errno.h>
int main(int argc, char **argv) {
    const char *path = argc > 1 ? argv[1] : ".";
    pid_t pid = fork();
    if (pid < 0) {
        perror("fork");
        return 1;
    }
    if (pid == 0) {
        printf("Child PID: %ld\n", (long)getpid());
        struct stat st;
        if (stat(path, &st) == 0)
        printf("stat: %s is %s\n", path, S_ISDIR(st.st_mode) ? "directory" : "not a directory");
        else perror("stat");
        DIR *d = opendir(path);
        if (!d) {
            perror("opendir");
            _exit(1);
        }
        struct dirent *entry;
        puts("Directory entries:");
        while ((entry = readdir(d)) != NULL) puts(entry->d_name);
        closedir(d);
        int fd = open("/dev/null", O_RDONLY);
        if (fd >= 0) close(fd);
        execlp("echo", "echo", "exec() successfully ran echo", (char *)NULL);
        perror("execlp");
        _exit(1);
    }
    printf("Parent PID: %ld; child PID: %ld\n", (long)getpid(), (long)pid);
    int status;
    if (waitpid(pid, &status, 0) < 0) {
        perror("waitpid");
        return 1;
    }
    printf("Child exit status: %d\n", WIFEXITED(status) ? WEXITSTATUS(status) : -1);
    return 0;
}

#include <stdio.h>
typedef struct {
    int id, burst, priority, waiting, turnaround;
}
Process;
int main(void) {
    Process p[100], t;
    int n;
    printf("Number of processes (1-100): ");
    if (scanf("%d", &n) != 1 || n < 1 || n > 100) return puts("Invalid count."), 1;
    for (int i=0;i<n;i++) {
        p[i].id=i+1;
        printf("P%d burst time and priority: ",i+1);
        if (scanf("%d%d",&p[i].burst,&p[i].priority)!=2 || p[i].burst<0) return puts("Invalid input."),1;
    }
    for(int i=0;i<n-1;i++) for(int j=i+1;j<n;j++)
    if(p[j].priority<p[i].priority) {
        t=p[i];
        p[i]=p[j];
        p[j]=t;
    }
    int time=0, sw=0, st=0;
    puts("\nProcess  Priority  Burst  Waiting  Turnaround");
    for(int i=0;i<n;i++) {
        p[i].waiting=time;
        p[i].turnaround=time+p[i].burst;
        time+=p[i].burst;
        sw+=p[i].waiting;
        st+=p[i].turnaround;
        printf("P%d\t%d\t%d\t%d\t%d\n",p[i].id,p[i].priority,p[i].burst,p[i].waiting,p[i].turnaround);
    }
    printf("Average waiting: %.2f\nAverage turnaround: %.2f\n",(double)sw/n,(double)st/n);
    return 0;
}

CPU Scheduling — Round Robin

CCompile: gcc program.c -o program && ./program
#include <stdio.h>
int main(void) {
    int n,bt[100],rem[100],wt[100]= {
        0
    },tat[100],q,done,time=0,tw=0,tt=0;
    printf("Number of processes (1-100): ");
    if(scanf("%d",&n)!=1||n<1||n>100)return puts("Invalid count."),1;
    for(int i=0;i<n;i++) {
        printf("P%d burst time: ",i+1);
        if(scanf("%d",&bt[i])!=1||bt[i]<0)return puts("Invalid burst time."),1;
        rem[i]=bt[i];
    }
    printf("Time quantum: ");
    if(scanf("%d",&q)!=1||q<=0)return puts("Quantum must be positive."),1;
    do {
        done=1;
        for(int i=0;i<n;i++)if(rem[i]>0) {
            done=0;
            int run=rem[i]<q?rem[i]:q;
            time+=run;
            rem[i]-=run;
            if(rem[i]==0)tat[i]=time;
        }
    }
    while(!done);
    puts("\nProcess  Burst  Waiting  Turnaround");
    for(int i=0;i<n;i++) {
        wt[i]=tat[i]-bt[i];
        tw+=wt[i];
        tt+=tat[i];
        printf("P%d\t%d\t%d\t%d\n",i+1,bt[i],wt[i],tat[i]);
    }
    printf("Average waiting: %.2f\nAverage turnaround: %.2f\n",(double)tw/n,(double)tt/n);
    return 0;
}
EX. 3

Semaphore — Producer/Consumer (POSIX threads)

CCompile: gcc program.c -o program && ./program
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#define SIZE 5
#define ITEMS 10
int buffer[SIZE], in=0, out=0;
sem_t empty_slots, full_slots;
pthread_mutex_t mutex=PTHREAD_MUTEX_INITIALIZER;
void *producer(void *arg) {
    for(int i=1;i<=ITEMS;i++) {
        sem_wait(&empty_slots);
        pthread_mutex_lock(&mutex);
        buffer[in]=i;
        in=(in+1)%SIZE;
        printf("Produced %d\n",i);
        pthread_mutex_unlock(&mutex);
        sem_post(&full_slots);
    }
    return NULL;
}
void *consumer(void *arg) {
    for(int i=1;i<=ITEMS;i++) {
        sem_wait(&full_slots);
        pthread_mutex_lock(&mutex);
        int item=buffer[out];
        out=(out+1)%SIZE;
        printf("Consumed %d\n",item);
        pthread_mutex_unlock(&mutex);
        sem_post(&empty_slots);
    }
    return NULL;
}
int main(void) {
    pthread_t p,c;
    sem_init(&empty_slots,0,SIZE);
    sem_init(&full_slots,0,0);
    if(pthread_create(&p,NULL,producer,NULL)||pthread_create(&c,NULL,consumer,NULL))return perror("pthread_create"),1;
    pthread_join(p,NULL);
    pthread_join(c,NULL);
    sem_destroy(&empty_slots);
    sem_destroy(&full_slots);
    pthread_mutex_destroy(&mutex);
    return 0;
}

Shared Memory — System V IPC (writer/reader demo)

CCompile: gcc program.c -o program && ./program
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/shm.h>
#include <sys/wait.h>
#include <unistd.h>
#define SHM_SIZE 1024
int main(void) {
    int id=shmget(IPC_PRIVATE,SHM_SIZE,IPC_CREAT|0600);
    if(id<0)return perror("shmget"),1;
    char *mem=shmat(id,NULL,0);
    if(mem==(void*)-1) {
        perror("shmat");
        shmctl(id,IPC_RMID,NULL);
        return 1;
    }
    pid_t pid=fork();
    if(pid<0) {
        perror("fork");
        shmdt(mem);
        shmctl(id,IPC_RMID,NULL);
        return 1;
    }
    if(pid==0) {
        snprintf(mem,SHM_SIZE,"Hello from child process %ld",(long)getpid());
        shmdt(mem);
        _exit(0);
    }
    waitpid(pid,NULL,0);
    printf("Shared memory says: %s\n",mem);
    shmdt(mem);
    shmctl(id,IPC_RMID,NULL);
    return 0;
}

Banker's Algorithm — Deadlock Avoidance / Safe Sequence

CCompile: gcc program.c -o program && ./program
#include <stdio.h>
#define P 10
#define R 10
int main(void) {
    int n,m,alloc[P][R],max[P][R],need[P][R],avail[R],finish[P]= {
        0
    },seq[P],count=0;
    printf("Processes and resource types (max 10 each): ");
    if(scanf("%d%d",&n,&m)!=2||n<1||n>P||m<1||m>R)return puts("Invalid dimensions."),1;
    puts("Enter allocation matrix:");
    for(int i=0;i<n;i++)for(int j=0;j<m;j++)if(scanf("%d",&alloc[i][j])!=1||alloc[i][j]<0)return 1;
    puts("Enter maximum-need matrix:");
    for(int i=0;i<n;i++)for(int j=0;j<m;j++)if(scanf("%d",&max[i][j])!=1||max[i][j]<alloc[i][j])return puts("Max must be >= allocation."),1;
    puts("Enter available resources:");
    for(int j=0;j<m;j++)if(scanf("%d",&avail[j])!=1||avail[j]<0)return 1;
    for(int i=0;i<n;i++)for(int j=0;j<m;j++)need[i][j]=max[i][j]-alloc[i][j];
    while(count<n) {
        int progress=0;
        for(int i=0;i<n;i++)if(!finish[i]) {
            int can=1;
            for(int j=0;j<m;j++)if(need[i][j]>avail[j])can=0;
            if(can) {
                for(int j=0;j<m;j++)avail[j]+=alloc[i][j];
                seq[count++]=i;
                finish[i]=1;
                progress=1;
            }
        }
        if(!progress)break;
    }
    if(count==n) {
        printf("Safe sequence: ");
        for(int i=0;i<n;i++)printf("P%d%s",seq[i],i==n-1?"\n":" -> ");
    }
    else puts("Unsafe state: no safe sequence exists.");
    return 0;
}

    puts("2. Hold and wait: request all required resources together, or release held resources before requesting more.");

Memory Allocation — First Fit

CCompile: gcc program.c -o program && ./program
#include <stdio.h>
int main(void) {
    int nb,np,b[100],p[100],a[100];
    printf("Number of blocks (1-100): ");
    if(scanf("%d",&nb)!=1||nb<1||nb>100)return 1;
    for(int i=0;i<nb;i++) {
        printf("Block %d size: ",i+1);
        if(scanf("%d",&b[i])!=1||b[i]<0)return 1;
    }
    printf("Number of processes (1-100): ");
    if(scanf("%d",&np)!=1||np<1||np>100)return 1;
    for(int i=0;i<np;i++) {
        printf("Process %d size: ",i+1);
        if(scanf("%d",&p[i])!=1||p[i]<0)return 1;
        a[i]=-1;
    }
    for(int i=0;i<np;i++)for(int j=0;j<nb;j++)if(b[j]>=p[i]) {
        a[i]=j;
        b[j]-=p[i];
        break;
    }
    puts("\nProcess  Size  Block (remaining capacity tracked)");
    for(int i=0;i<np;i++)if(a[i]>=0)printf("P%d\t%d\t%d\n",i+1,p[i],a[i]+1);
    else printf("P%d\t%d\tNot allocated\n",i+1,p[i]);
    return 0;
}

Page Replacement — FIFO

CCompile: gcc program.c -o program && ./program
#include <stdio.h>
int main(void) {
    int n,f,p[1000],frame[100],faults=0,next=0;
    printf("Number of page references (1-1000): ");
    if(scanf("%d",&n)!=1||n<1||n>1000)return 1;
    printf("Reference string: ");
    for(int i=0;i<n;i++)if(scanf("%d",&p[i])!=1)return 1;
    printf("Number of frames (1-100): ");
    if(scanf("%d",&f)!=1||f<1||f>100)return 1;
    for(int j=0;j<f;j++)frame[j]=-1;
    for(int i=0;i<n;i++) {
        int hit=0;
        for(int j=0;j<f;j++)if(frame[j]==p[i])hit=1;
        if(!hit) {
            frame[next]=p[i];
            next=(next+1)%f;
            faults++;
        }
        printf("%d: ",p[i]);
        for(int j=0;j<f;j++)if(frame[j]<0)printf("- ");
        else printf("%d ",frame[j]);
        puts(hit?"(hit)":"(fault)");
    }
    printf("Total page faults: %d\n",faults);
    return 0;
}

Page Replacement — LRU

#include <stdio.h>
int main(void) {
    int n,f,p[1000],frame[100],last[100],faults=0,tick=0;
    printf("Number of page references (1-1000): ");
    if(scanf("%d",&n)!=1||n<1||n>1000)return 1;
    printf("Reference string: ");
    for(int i=0;i<n;i++)if(scanf("%d",&p[i])!=1)return 1;
    printf("Number of frames (1-100): ");
    if(scanf("%d",&f)!=1||f<1||f>100)return 1;
    for(int j=0;j<f;j++) {
        frame[j]=-1;
        last[j]=0;
    }
    for(int i=0;i<n;i++) {
        tick++;
        int pos=-1;
        for(int j=0;j<f;j++)if(frame[j]==p[i])pos=j;
        if(pos>=0)last[pos]=tick;
        else {
            faults++;
            pos=-1;
            for(int j=0;j<f;j++)if(frame[j]==-1) {
                pos=j;
                break;
            }
            if(pos<0) {
                pos=0;
                for(int j=1;j<f;j++)if(last[j]<last[pos])pos=j;
            }
            frame[pos]=p[i];
            last[pos]=tick;
        }
        printf("%d: ",p[i]);
        for(int j=0;j<f;j++)if(frame[j]<0)printf("- ");
        else printf("%d ",frame[j]);
        puts(pos>=0?"":"");
    }
    printf("Total page faults: %d\n",faults);
    return 0;
}

Security Model — Bell–LaPadula (confidentiality)

CCompile: gcc program.c -o program && ./program
#include <stdio.h>
typedef enum {
    PUBLIC=0, CONFIDENTIAL=1, SECRET=2, TOP_SECRET=3
}
Level;
const char *name(Level l) {
    const char *n[]= {
        "Public","Confidential","Secret","Top Secret"
    };
    return n[l];
}
int main(void) {
    int s,o,read,write;
    printf("Subject clearance (0 Public, 1 Confidential, 2 Secret, 3 Top Secret): ");
    if(scanf("%d",&s)!=1||s<0||s>3)return 1;
    printf("Object classification (0-3): ");
    if(scanf("%d",&o)!=1||o<0||o>3)return 1;
    printf("Operation (1=read, 2=write): ");
